--
Create an Azure Container Registry (ACR)
az acr create --resource-group veloraResourceGroup --name veloradevacr  --sku Basic

--
Enable Admin Account for ACR Pull
az acr update -n veloradevacr  --admin-enabled true

--
Log in to the container registry
az acr login --name veloradevacr 
Login Succeeded

--
Tag a container image

get the login server address
az acr list --resource-group veloraResourceGroup --query "[].{acrLoginServer:loginServer}" --output table
veloradevacr.azurecr.io

--
Tag your images

docker tag veloraapi:latest veloradevacr.azurecr.io/veloraapi:v1
docker tag veloraclient:latest veloradevacr.azurecr.io/veloraclient:v1


Check
docker images

docker push veloradevacr.azurecr.io/veloraclient:v1

Create AKS cluster with attaching ACR
#az aks create --resource-group veloraResourceGroup --name veloraAKSCluster --node-count 1 --generate-ssh-keys --attach-acr veloradevacr 


az aks create `
    --resource-group veloraResourceGroup `
    --name veloraAKSCluster `
    --location denmarkeast `
    --node-count 1 `
    --node-vm-size Standard_F4s_v2 `
    --tier free `
    --enable-managed-identity `
    --os-sku Ubuntu `
    --node-osdisk-type Managed `
    --node-osdisk-size 32 `
    --generate-ssh-keys

--
Install the Kubernetes CLI
az aks install-cli

--
Stop AKS
az aks stop --resource-group veloraResourceGroup --name veloraAKSCluster

This stops your AKS cluster and its worker node, suspending VM compute charges.


Start AKS
az aks start --resource-group veloraResourceGroup --name veloraAKSCluster
Your Kubernetes Deployments, Services and configuration are preserved.

Check cluster status

az aks show -g veloraResourceGroup -n veloraAKSCluster --query "{Name:name,PowerState:powerState.code,ProvisioningState:provisioningState}" -o table

--
Clean All AKS and Azure Resources

az group delete --name veloraResourceGroup --yes --no-wait
